# XPMultiCrew's bundled generic CSL package (see docs/plan.md section 5 and
# the "generic CSL now, richer packages optional later" decision).
#
# A single hand-built placeholder model (fuselage/wing/tail boxes, flat
# orange-red texture - see generic.obj/generic.png) used for every remote
# aircraft, regardless of its real ICAO type, so Formation mode works with
# zero manual setup. Format per the OBJ8 CSL spec used by XPMP2/libxplanemp.

EXPORT_NAME XPMultiCrew_Generic

OBJ8_AIRCRAFT Generic

OBJ8 SOLID YES XPMultiCrew_Generic/generic.obj

# VERT_OFFSET follows the normal XPMP2 meaning: how far to lift the model
# so its lowest point sits at the altitude it's given, which is the bottom
# of the peer's landing gear (the plugin subtracts the peer's reported
# reference-point height, see csl_aircraft.cpp). generic.obj's lowest
# vertex is at Y=0.05, so it just needs to come down those 5cm.
#
# Set explicitly because XPMP2's auto-detect (FetchVertOfsFromObjFile)
# gets this mesh wrong: with every vertex above Y=0 it falls back to the
# negated TOP of the mesh (-1.45). Up to v0.2.2 this line was -2.5, which
# also compensated for the reference-point height the plugin didn't send
# back then.
VERT_OFFSET -0.05

ICAO GENR
